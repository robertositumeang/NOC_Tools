function doPost(e) { try { var data = JSON.parse(e.postData.contents); var rows = data.rows || []; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Database TT"); if (!sheet) return ContentService.createTextOutput( JSON.stringify({ok:false,error:"Sheet 'Database TT' tidak ditemukan"})) .setMimeType(ContentService.MimeType.JSON); // Find last row with actual data in column C (Site ID), // skipping formula rows (#N/A) below the real data. var totalRows = sheet.getLastRow(); var lastDataRow = 1; // default to row 1 if sheet is empty if (totalRows > 1) { var colCVals = sheet.getRange(2, 3, totalRows - 1, 1).getValues(); for (var i = 0; i < colCVals.length; i++) { var cell = colCVals[i][0]; if (cell !== "" && cell !== null && typeof cell === "string" && cell.toString().indexOf("#") !== 0) { lastDataRow = i + 2; // +2 because we started from row 2 } } } var startRow = lastDataRow + 1; // Kolom yang akan diisi (1-indexed) var COL = { siteId:3, alarmId:13, alarmTitle:14, ticketId:15, ticketDur:16, stateCond:17, dateDown:19, status:24, admState1:25, admState2:26, opState:27 }; rows.forEach(function(row, i) { var r = startRow + i; sheet.getRange(r, COL.siteId).setValue(row["Site ID"] || ""); sheet.getRange(r, COL.alarmId).setValue(row["Alarm ID"] || ""); sheet.getRange(r, COL.alarmTitle).setValue(row["Alarm Title"] || ""); sheet.getRange(r, COL.ticketId).setValue(row["Ticket ID"] || ""); sheet.getRange(r, COL.ticketDur).setValue(row["Ticket Duration"] || ""); sheet.getRange(r, COL.stateCond).setValue(row["State Condition"] || ""); sheet.getRange(r, COL.dateDown).setValue(row["Date Time Site Down"] || ""); sheet.getRange(r, COL.status).setValue(row["Status"] || ""); sheet.getRange(r, COL.admState1).setValue(row["Administrative State 1"] || ""); sheet.getRange(r, COL.admState2).setValue(row["Administrative State 2"] || ""); sheet.getRange(r, COL.opState).setValue(row["Operation State"] || ""); }); return ContentService.createTextOutput( JSON.stringify({ok:true,written:rows.length,startRow:startRow})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput( JSON.stringify({ok:false,error:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
