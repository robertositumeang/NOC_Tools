Konfigurasi Google Apps Script untuk menulis data langsung ke spreadsheet
Jika masih menulis jauh di bawah sitelist, perbarui kode Apps Script dan deploy sebagai versi baru.
function doPost(e) { try { var data = JSON.parse(e.postData.contents); var rows = data.rows || []; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Database TT"); if (!sheet) return ContentService.createTextOutput( JSON.stringify({ok:false,error:"Sheet 'Database TT' tidak ditemukan"})) .setMimeType(ContentService.MimeType.JSON); var lastDataRow = 1; var totalRows = sheet.getLastRow(); if (totalRows > 1) { var siteIds = sheet.getRange(2, 3, totalRows - 1, 1).getDisplayValues(); for (var i = siteIds.length - 1; i >= 0; i--) { var siteId = siteIds[i][0].trim(); if (siteId !== "" && siteId.charAt(0) !== "#") { lastDataRow = i + 2; break; } } } var startRow = lastDataRow + 1; var COL = { siteId:3, alarmId:13, alarmTitle:14, ticketId:15, ticketDur:16, stateCond:17, dateDown:19, status:24, admState1:25, admState2:26, opState:27 }; rows.forEach(function(row, i) { var r = startRow + i; sheet.getRange(r, COL.siteId).setValue(row["Site ID"] || ""); sheet.getRange(r, COL.alarmId).setValue(row["Alarm ID"] || ""); sheet.getRange(r, COL.alarmTitle).setValue(row["Alarm Title"] || ""); sheet.getRange(r, COL.ticketId).setValue(row["Ticket ID"] || ""); sheet.getRange(r, COL.ticketDur).setValue(row["Ticket Duration"] || ""); sheet.getRange(r, COL.stateCond).setValue(row["State Condition"] || ""); sheet.getRange(r, COL.dateDown).setValue(row["Date Time Site Down"] || ""); sheet.getRange(r, COL.status).setValue(row["Status"] || ""); sheet.getRange(r, COL.admState1).setValue(row["Administrative State 1"] || ""); sheet.getRange(r, COL.admState2).setValue(row["Administrative State 2"] || ""); sheet.getRange(r, COL.opState).setValue(row["Operation State"] || ""); }); return ContentService.createTextOutput( JSON.stringify({ok:true,written:rows.length,startRow:startRow})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput( JSON.stringify({ok:false,error:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
